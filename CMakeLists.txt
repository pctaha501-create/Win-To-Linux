cmake_minimum_required(VERSION 3.21)
project(WinToLinux VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Network Qml Quick QuickControls2)
add_subdirectory(src/core)
add_subdirectory(src/app)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS win-to-linux RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/start-isolated-preview.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME win-to-linux-isolated-preview)
install(PROGRAMS scripts/start-standalone-session.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME win-to-linux-standalone-session)
install(PROGRAMS scripts/capture-xwayland-env.sh DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/linux/win-to-linux.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
